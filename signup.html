window.signup = async function () { const name = document.getElementById("name").value; const phone = document.getElementById("phone").value; const password = document.getElementById("password").value; const msg = document.getElementById("msg"); msg.innerText = "جاري إنشاء الحساب..."; try { // Supabase Auth const { data, error } = await supabase.auth.signUp({ email: phone + "@phone.com", password }); if(error){ msg.innerText = error.message; return; } // Database insert const { error: dbError } = await supabase .from("students") .insert([ { full_name: name, phone: phone, role: "student", activated: false } ]); if(dbError){ msg.innerText = dbError.message; return; } msg.style.color = "green"; msg.innerText = "تم إنشاء الحساب بنجاح"; setTimeout(()=>{ window.location.href = "login.html"; },1500); } catch(err){ console.log(err); msg.innerText = "حدث خطأ"; } }
